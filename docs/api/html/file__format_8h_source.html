226static inline const Section *fileinfo_find_section(const FileInfo *fi,
227 const char *name)
228{
229 if (!fi || !name) return NULL;
230 for (uint32_t i = 0; i < fi->section_count; i++) {
231 if (fi->sections[i].name &&
232 strcmp(fi->sections[i].name, name) == 0) {

234 }

236 return NULL;
